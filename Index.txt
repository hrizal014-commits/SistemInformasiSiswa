        const allowedMenus = rolePermissions[currentUserRole] || [];
        const isAll = allowedMenus.includes('all');

        document.querySelectorAll('.access-node').forEach(btn => {
            const menuId = btn.getAttribute('data-menu');
            if (isAll || allowedMenus.includes(menuId)) btn.style.display = 'flex';
            else btn.style.display = 'none';
        });

        document.querySelectorAll('.menu-group').forEach(group => {
            if(group.id === 'menuPengaturan') return;
            const accessList = group.getAttribute('data-access').split(',');
            const hasAccess = isAll || accessList.some(item => allowedMenus.includes(item));
            group.style.display = hasAccess ? 'block' : 'none';
        });

        document.getElementById('menuPengaturan').style.display = (currentUserRole === 'admin') ? 'block' : 'none';
        
        // Pindah ke Dashboard
        document.getElementById('nav-dashboard').click();
    }

    function applyFiltersSiswaAktif() {
        let fKelas = document.getElementById('filterKelasAktif').value.toLowerCase();
        let fMitra = document.getElementById('filterMitraAktif').value.toLowerCase();
        let fRombel = document.getElementById('filterRombelAktif').value.toLowerCase();
        let fSearch = document.getElementById('searchInput').value.toLowerCase();
        let tbody = document.getElementById('tabelSiswaBody');

        let filtered = globalDataAktif.filter(s => {
            let mKelas = fKelas === "" || (s.kelas||"").toLowerCase() === fKelas;
            let mMitra = fMitra === "" || (s.asalMitra||"").toLowerCase() === fMitra;
            let mRombel = fRombel === "" || (s.rombel||"").toLowerCase() === fRombel;
            let mSearch = fSearch === "" || (s.nama||"").toLowerCase().includes(fSearch) || (s.nis||"").toString().includes(fSearch);
            return mKelas && mMitra && mRombel && mSearch;
        });

        tbody.innerHTML = filtered.length ? filtered.map((s, i) => buildRowHTML(s, i, 'aktif')).join('') : '<tr><td colspan="7" class="text-center py-4 text-muted">Tidak ada data siswa yang cocok dengan filter.</td></tr>';
    }

    function simpanKesiswaanManual(event, tipe) {
       event.preventDefault();
       let form = event.target;
       let nisSelect = document.getElementById(`${tipe}Nis`).value;
       
       if(!nisSelect) { Swal.fire('Error', 'Pilih nama siswa terlebih dahulu!', 'warning'); return; }
       let siswaObj = globalDataAktif.find(s => s.nis == nisSelect);
       let newData = { id: Date.now(), nis: nisSelect, nama: siswaObj.nama, kelas: siswaObj.kelas };

       if(tipe === 'prestasi') {
         newData.prestasi = document.getElementById('prestasiNama').value; newData.tingkat = document.getElementById('prestasiTingkat').value; newData.tanggal = document.getElementById('prestasiTanggal').value;
         dataPrestasi.push(newData); renderTabelKesiswaan('Prestasi', dataPrestasi);
       } else if(tipe === 'pelanggaran') {
         newData.pelanggaran = document.getElementById('pelanggaranBentuk').value; newData.poin = document.getElementById('pelanggaranPoin').value; newData.tanggal = document.getElementById('pelanggaranTanggal').value; newData.tindak = document.getElementById('pelanggaranTindak').value;
         dataPelanggaran.push(newData); renderTabelKesiswaan('Pelanggaran', dataPelanggaran);
       } else if(tipe === 'bakatminat') { // Tambahan logika bakat minat
         newData.bakat = document.getElementById('bakatminatBakat').value; newData.minat = document.getElementById('bakatminatMinat').value; newData.keterangan = document.getElementById('bakatminatKet').value;
         dataBakatMinat.push(newData); renderTabelKesiswaan('BakatMinat', dataBakatMinat);
       } else if(tipe === 'keaktifan') {
         newData.organisasi = document.getElementById('keaktifanNama').value; newData.jabatan = document.getElementById('keaktifanJabatan').value;
         dataKeaktifan.push(newData); renderTabelKesiswaan('Keaktifan', dataKeaktifan);
       } else if(tipe === 'partisipasi') {
         newData.kegiatan = document.getElementById('partisipasiKegiatan').value; newData.peran = document.getElementById('partisipasiPeran').value; newData.tanggal = document.getElementById('partisipasiTanggal').value;
         dataPartisipasi.push(newData); renderTabelKesiswaan('Partisipasi', dataPartisipasi);
       } else if(tipe === 'keuangan') {
         newData.jenis = document.getElementById('keuanganJenis').value; newData.nominal = document.getElementById('keuanganNominal').value; newData.tanggal = document.getElementById('keuanganTanggal').value;
         dataKeuangan.push(newData); renderTabelKesiswaan('Keuangan', dataKeuangan);
       }

       Swal.fire({ icon: 'success', title: 'Berhasil Tersimpan', timer: 1500, showConfirmButton: false});
       form.reset();
       $('.dropdownSiswaAktif').val(null).trigger('change');
       
       if(typeof google !== 'undefined') google.script.run.simpanKesiswaanKeDB({tipe: tipe, data: newData});
    }

    function hapusKesiswaan(id, tipe) {
    Swal.fire({ title: 'Hapus data?', icon: 'warning', showCancelButton: true }).then(r => {
        if(r.isConfirmed) {
            // Hapus dari visual UI 
            if(tipe === 'Prestasi') { dataPrestasi = dataPrestasi.filter(d => d.id != id); renderTabelKesiswaan(tipe, dataPrestasi); }
            if(tipe === 'Pelanggaran') { dataPelanggaran = dataPelanggaran.filter(d => d.id != id); renderTabelKesiswaan(tipe, dataPelanggaran); }
            if(tipe === 'BakatMinat') { dataBakatMinat = dataBakatMinat.filter(d => d.id != id); renderTabelKesiswaan(tipe, dataBakatMinat); }
            if(tipe === 'Keaktifan') { dataKeaktifan = dataKeaktifan.filter(d => d.id != id); renderTabelKesiswaan(tipe, dataKeaktifan); }
            if(tipe === 'Partisipasi') { dataPartisipasi = dataPartisipasi.filter(d => d.id != id); renderTabelKesiswaan(tipe, dataPartisipasi); }
            if(tipe === 'Keuangan') { dataKeuangan = dataKeuangan.filter(d => d.id != id); renderTabelKesiswaan(tipe, dataKeuangan); }
            
            // Eksekusi Hapus permanen di Database Google Sheets
            if(typeof google !== 'undefined') {
                google.script.run.hapusKesiswaanDariDB(id, tipe);
            }
        }
    });
}

    function renderTabelKesiswaan(tipe, arrData) {
       const tbody = document.getElementById(`tabel${tipe}Body`);
       if(arrData.length === 0) { tbody.innerHTML = `<tr><td colspan="7" class="text-center py-4 text-muted">Belum ada data.</td></tr>`; return; }
       
       let html = arrData.map((d, i) => {
          let cols = '';
          if(tipe === 'Prestasi') cols = `<td>${d.kelas}</td><td>${d.prestasi}</td><td>${d.tingkat}</td><td>${d.tanggal}</td>`;
          if(tipe === 'Pelanggaran') cols = `<td>${d.pelanggaran}</td><td>${d.tindak}</td><td><span class="badge bg-danger">-${d.poin}</span></td><td>${d.tanggal}</td>`;
          if(tipe === 'BakatMinat') cols = `<td>${d.kelas}</td><td>${d.bakat}</td><td>${d.minat}</td><td>${d.keterangan || '-'}</td>`;
          if(tipe === 'Keaktifan') cols = `<td>${d.kelas}</td><td>${d.organisasi}</td><td>${d.jabatan}</td>`;
          if(tipe === 'Partisipasi') cols = `<td>${d.kegiatan}</td><td>${d.peran}</td><td>${d.tanggal}</td>`;
          if(tipe === 'Keuangan') cols = `<td>${d.kelas}</td><td>${d.jenis}</td><td>Rp. ${parseInt(d.nominal).toLocaleString('id-ID')}</td><td>${d.tanggal}</td>`;
          return `<tr><td>${i+1}</td><td class="fw-bold text-primary">${d.nama}</td>${cols}<td class="text-center col-aksi no-print"><button class="btn-action bg-danger text-white" onclick="hapusKesiswaan(${d.id}, '${tipe}')"><i class="fas fa-trash"></i></button></td></tr>`;
       }).join('');
       tbody.innerHTML = html;
    }

    function prosesMasukRombel() {
       let namaRombelBaru = document.getElementById('pilihRombelTujuan').value;
       let selectedCheckboxes = Array.from(document.querySelectorAll('.checkSiswaRombel:checked'));
       if(!namaRombelBaru) { Swal.fire('Gagal', 'Pilih Rombel Tujuan terlebih dahulu!', 'warning'); return; }
       if(selectedCheckboxes.length === 0) { Swal.fire('Gagal', 'Centang minimal satu siswa untuk dimasukkan!', 'warning'); return; }

       Swal.fire({ title: 'Konfirmasi', text: `Masukan ${selectedCheckboxes.length} siswa ke rombel ${namaRombelBaru}?`, icon: 'question', showCancelButton: true }).then(res => {
          if(res.isConfirmed) {
             let rowIndexArray = selectedCheckboxes.map(cb => cb.value);
             if(typeof google === 'undefined') {
                globalDataAktif.forEach(s => { if(rowIndexArray.includes(s.rowIndex.toString())) s.rombel = namaRombelBaru; });
                Swal.fire('Berhasil', 'Siswa berhasil dimasukkan ke rombel (Lokal)', 'success');
                renderSiswaRombel(); applyFiltersSiswaAktif(); return;
             }
             document.getElementById('loadingOverlay').style.display = 'flex';
             google.script.run.withSuccessHandler(resServer => {
                document.getElementById('loadingOverlay').style.display = 'none';
                Swal.fire('Berhasil', `Data sinkron!`, 'success');
                loadDataSiswa();
             }).assignSiswaKeRombel(rowIndexArray, namaRombelBaru);
          }
       });
    }

    function hitungStatistikMitra() {
        let mitraMap = {};
        globalDataAktif.forEach(s => {
            let namaMitra = (s.asalMitra && s.asalMitra.trim() !== "") ? s.asalMitra.trim() : "Tidak Diketahui / Kosong";
            if(!mitraMap[namaMitra]) { mitraMap[namaMitra] = { L: 0, P: 0, Total: 0, X_L: 0, X_P: 0, X_Total: 0, XI_L: 0, XI_P: 0, XI_Total: 0, XII_L: 0, XII_P: 0, XII_Total: 0 }; }
            mitraMap[namaMitra].Total++; if(s.jk === 'L') mitraMap[namaMitra].L++; if(s.jk === 'P') mitraMap[namaMitra].P++;
            if(s.kelas === 'X') { mitraMap[namaMitra].X_Total++; if(s.jk === 'L') mitraMap[namaMitra].X_L++; if(s.jk === 'P') mitraMap[namaMitra].X_P++; } 
            else if(s.kelas === 'XI') { mitraMap[namaMitra].XI_Total++; if(s.jk === 'L') mitraMap[namaMitra].XI_L++; if(s.jk === 'P') mitraMap[namaMitra].XI_P++; } 
            else if(s.kelas === 'XII') { mitraMap[namaMitra].XII_Total++; if(s.jk === 'L') mitraMap[namaMitra].XII_L++; if(s.jk === 'P') mitraMap[namaMitra].XII_P++; }
        });

        let htmlRows = ''; let t = { L: 0, P: 0, Total: 0, X_L: 0, X_P: 0, X_Total: 0, XI_L: 0, XI_P: 0, XI_Total: 0, XII_L: 0, XII_P: 0, XII_Total: 0 };
        for(let mitra in mitraMap) {
            let d = mitraMap[mitra];
            t.L+=d.L; t.P+=d.P; t.Total+=d.Total; t.X_L+=d.X_L; t.X_P+=d.X_P; t.X_Total+=d.X_Total; t.XI_L+=d.XI_L; t.XI_P+=d.XI_P; t.XI_Total+=d.XI_Total; t.XII_L+=d.XII_L; t.XII_P+=d.XII_P; t.XII_Total+=d.XII_Total;
            htmlRows += `<tr><td class="text-start fw-bold text-primary">${mitra}</td><td>${d.X_L}</td><td>${d.X_P}</td><td class="fw-bold text-success bg-light">${d.X_Total}</td><td>${d.XI_L}</td><td>${d.XI_P}</td><td class="fw-bold text-success bg-light">${d.XI_Total}</td><td>${d.XII_L}</td><td>${d.XII_P}</td><td class="fw-bold text-success bg-light">${d.XII_Total}</td><td class="text-primary">${d.L}</td><td class="text-danger">${d.P}</td><td class="fw-bold bg-primary text-white">${d.Total}</td></tr>`;
        }
        if(htmlRows === '') htmlRows = '<tr><td colspan="13" class="text-muted">Tidak ada data aktif</td></tr>';

    function populateDropdownFilters() {
        let mitraSet = new Set();
        globalDataAktif.forEach(s => { if (s.asalMitra && s.asalMitra.trim() !== "") mitraSet.add(s.asalMitra.trim()); });
        let opsiMitra = '<option value="">Semua Mitra</option>';
        Array.from(mitraSet).sort().forEach(mitra => { opsiMitra += `<option value="${mitra}">${mitra}</option>`; });
        document.getElementById('filterMitraAktif').innerHTML = opsiMitra;
    }

    function loadDataSiswa() {
      if(typeof google === 'undefined') return;
      const tbody = document.getElementById('tabelSiswaBody'); tbody.innerHTML = '<tr><td colspan="7" class="text-center py-4">Memuat data...</td></tr>';
      google.script.run.withSuccessHandler(data => {
        document.getElementById('loadingOverlay').style.display = 'none';
        globalDataAktif = (data||[]).filter(s => s.status === 'Aktif' || !s.status);
        
        // Auto Sort Abjad
        globalDataAktif.sort((a, b) => (a.nama || '').localeCompare(b.nama || ''));

        applyFiltersSiswaAktif();
        hitungStatistikMitra();
        renderTabelKenaikan();
        renderTabelKelulusan();
        populateDropdownFilters();
        initSelect2CariNama();
      }).getAllDataSiswa();
    }

    function loadDataAlumni() {
      if(typeof google === 'undefined') return;
      document.getElementById('loadingOverlay').style.display = 'flex';
      const tbody = document.getElementById('tabelAlumniBody'); tbody.innerHTML = '<tr><td colspan="6" class="text-center py-4">Memuat data alumni...</td></tr>';

        document.getElementById('loadingOverlay').style.display = 'none';

    }

    function renderTabelKenaikan() {
        let filterKelas = document.getElementById('filterKelasNaik').value;
        let tbody = document.getElementById('tabelKenaikanBody');
        document.getElementById('checkAllNaik').checked = false;
        let filteredData = globalDataAktif.filter(s => s.kelas === filterKelas);

    }

    function renderTabelKelulusan() {
        let tbody = document.getElementById('tabelKelulusanBody');

    function toggleCheckAll(sourceId, classTarget) {

        document.querySelectorAll('.' + classTarget).forEach(cb => cb.checked = isChecked);
    }

    function prosesKenaikan() {
        let targetKelas = document.getElementById('targetKelasNaik').value;
        let selectedRows = Array.from(document.querySelectorAll('.checkNaik:checked')).map(cb => cb.value);
        if(selectedRows.length === 0) { Swal.fire('Perhatian', 'Pilih minimal satu siswa untuk dinaikkan kelasnya!', 'warning'); return; }
        Swal.fire({ title: 'Proses Kenaikan Kelas?', text: `${selectedRows.length} siswa akan dinaikkan ke Kelas ${targetKelas}.`, icon: 'question', showCancelButton: true }).then((r) => {
            if (r.isConfirmed) {
                if(typeof google === 'undefined') return;
                document.getElementById('loadingOverlay').style.display = 'flex';
                google.script.run.withSuccessHandler(res => {
                    document.getElementById('loadingOverlay').style.display = 'none';
                    if(res.status === 'success') { Swal.fire('Berhasil!', res.message, 'success'); loadDataSiswa(); } else Swal.fire('Gagal', res.message, 'error');
                }).prosesKenaikanMasal(selectedRows, targetKelas);
            }
        });
    }

    function prosesKelulusan() {
        let tahunLulus = document.getElementById('inputTahunLulus').value;
        let selectedRows = Array.from(document.querySelectorAll('.checkLulus:checked')).map(cb => cb.value);
        if(!tahunLulus) { Swal.fire('Perhatian', 'Harap isi Tahun Lulus!', 'warning'); return; }
        if(selectedRows.length === 0) { Swal.fire('Perhatian', 'Pilih minimal satu siswa untuk diluluskan!', 'warning'); return; }
        Swal.fire({ title: 'Proses Kelulusan?', text: `${selectedRows.length} siswa akan diubah statusnya menjadi Alumni.`, icon: 'warning', showCancelButton: true }).then((r) => {
            if (r.isConfirmed) {
                if(typeof google === 'undefined') return;
                document.getElementById('loadingOverlay').style.display = 'flex';
                google.script.run.withSuccessHandler(res => {
                    document.getElementById('loadingOverlay').style.display = 'none';
                    if(res.status === 'success') { Swal.fire('Berhasil!', res.message, 'success'); loadDataSiswa(); } else Swal.fire('Gagal', res.message, 'error');
                }).prosesKelulusanMasal(selectedRows, tahunLulus);
            }
        });
    }

    function buildRowHTML(siswa, index, type) {
      let id = siswa.rowIndex;
      if(type === 'aktif') {
        let jalurColor = siswa.jalurMasuk === 'Pindahan' ? 'bg-warning text-dark' : 'bg-primary text-white';
        let txtKelas = siswa.kelas || '-'; if(siswa.rombel) txtKelas += ` / <span class="badge bg-light text-dark border">${siswa.rombel}</span>`;
        return `<tr><td class="text-muted fw-bold">${index + 1}</td><td><div class="fw-bold">${siswa.nis||'-'}</div><small class="text-muted">${siswa.nisn||'-'}</small></td><td><div class="fw-bold text-primary">${siswa.nama||'-'}</div></td><td><span class="badge badge-custom bg-secondary">${txtKelas}</span></td><td><span class="badge badge-custom ${jalurColor}">${siswa.jalurMasuk||'-'}</span></td><td><div class="fw-bold">${siswa.tahunMasuk||'-'}</div></td><td class="text-center col-aksi no-print"><button class="btn-action bg-info text-white" onclick="lihatDetail('${id}', 'aktif')"><i class="fas fa-eye"></i></button> <button class="btn-action bg-warning text-dark" onclick="editData('${id}', 'aktif')"><i class="fas fa-edit"></i></button> <button class="btn-action bg-danger text-white" onclick="hapusData('${id}', 'aktif')"><i class="fas fa-trash"></i></button></td></tr>`;
      } else if(type === 'alumni') {
        return `<tr><td class="text-muted fw-bold">${index + 1}</td><td><div class="fw-bold">${siswa.nis||'-'}</div><small class="text-muted">${siswa.nisn||'-'}</small></td><td><div class="fw-bold text-primary">${siswa.nama||'-'}</div></td><td><span class="badge badge-custom bg-info text-dark">${siswa.tahunLulus || '-'}</span></td><td><small class="text-muted">${siswa.rombel||'-'}</small></td><td class="text-center col-aksi no-print"><button class="btn-action bg-info text-white" onclick="lihatDetail('${id}', 'alumni')"><i class="fas fa-eye"></i></button> <button class="btn-action bg-warning text-dark" onclick="editData('${id}', 'alumni')"><i class="fas fa-edit"></i></button> <button class="btn-action bg-danger text-white" onclick="hapusData('${id}', 'alumni')"><i class="fas fa-trash"></i></button></td></tr>`;
      } else {
        let sc = siswa.status === 'Pindah' ? 'bg-warning text-dark' : 'bg-danger';
        return `<tr><td class="text-muted fw-bold">${index + 1}</td><td><div class="fw-bold">${siswa.nis||'-'}</div></td><td><div class="fw-bold text-primary">${siswa.nama||'-'}</div></td><td><span class="badge badge-custom bg-secondary">${siswa.kelas||'-'}</span></td><td><span class="badge badge-custom ${sc}">${siswa.status||'-'}</span></td><td class="text-center col-aksi no-print"><button class="btn-action bg-info text-white" onclick="lihatDetail('${id}', 'nonaktif')"><i class="fas fa-eye"></i></button> <button class="btn-action bg-warning text-dark" onclick="editData('${id}', 'nonaktif')"><i class="fas fa-edit"></i></button> <button class="btn-action bg-danger text-white" onclick="hapusData('${id}', 'nonaktif')"><i class="fas fa-trash"></i></button></td></tr>`;
      }
    }

        if(tipe === 'aktif') return globalDataAktif.find(s => s.rowIndex == id);
        if(tipe === 'alumni') return globalDataAlumni.find(s => s.rowIndex == id);
        return globalDataNonAktif.find(s => s.rowIndex == id);
    }

    function editData(id, tipe) {
      const s = dapatkanData(id, tipe); if(!s) return;
      document.getElementById('editRowIndex').value = s.rowIndex; 
      ['asalMitra','nis','nisn','nama','jk','tempatLahir','tanggalLahir','kelas','rombel','asalSekolah','namaPanggilan','namaAyah','namaIbu','alamat','validEmis','status','tahunPelajaran','jalurMasuk','tahunMasuk','tahunLulus'].forEach(key => {
         if(document.getElementById(key)) document.getElementById(key).value = s[key] || '';
      });
      document.getElementById('btnSubmit').innerHTML = '<i class="fas fa-save me-2"></i>Update Data Siswa';
      document.getElementById('nav-form').click();
    }

    function hapusData(id, tipe) {
      const s = dapatkanData(id, tipe); if(!s) return;
      Swal.fire({ title: 'Hapus Data?', text: `Data ${s.nama} akan dihapus secara permanen!`, icon: 'warning', showCancelButton: true, confirmButtonColor: '#d33' }).then((r) => {
        if (r.isConfirmed) {
          if(typeof google === 'undefined') return;
          document.getElementById('loadingOverlay').style.display = 'flex';
          google.script.run.withSuccessHandler(res => {
            document.getElementById('loadingOverlay').style.display = 'none';
            if(res.status === 'success') { Swal.fire('Terhapus!', res.message, 'success'); if(tipe === 'aktif') loadDataSiswa(); else if(tipe === 'alumni') loadDataAlumni(); else loadDataNonAktif(); loadDashboardData(); } else Swal.fire('Error', res.message, 'error');
          }).hapusSiswa(s.rowIndex);
        }
      });
    }

      e.preventDefault();
      const btnSubmit = document.getElementById('btnSubmit'); const originalText = btnSubmit.innerHTML;
      btnSubmit.innerHTML = '<i class="fas fa-spinner fa-spin me-2"></i>Menyimpan...'; btnSubmit.disabled = true;

      const data = {};
      ['editRowIndex', 'asalMitra','nis','nisn','nama','jk','tempatLahir','tanggalLahir','kelas','rombel','asalSekolah','namaPanggilan','namaAyah','namaIbu','alamat','validEmis','status','tahunPelajaran','jalurMasuk','tahunMasuk','tahunLulus'].forEach(id => {
          let val = document.getElementById(id).value;
          data[id === 'editRowIndex' ? 'rowIndex' : id] = val;
      });
      
      if(typeof google === 'undefined') { btnSubmit.innerHTML = originalText; btnSubmit.disabled = false; Swal.fire('Info', 'Form tersimpan secara lokal (Tanpa DB)', 'info'); return; }

      google.script.run.withSuccessHandler(res => {
        btnSubmit.innerHTML = originalText; btnSubmit.disabled = false;
        if(res.status === "success") {
          Swal.fire({ icon: 'success', title: 'Berhasil!', text: res.message }).then(() => {
            document.getElementById('formSiswa').reset(); document.getElementById('editRowIndex').value = '';
            if(data.status === 'Aktif') document.getElementById('nav-dataSiswa').click(); else if(data.status === 'Alumni') document.getElementById('nav-alumni').click(); else document.getElementById('nav-nonAktif').click(); 
          });
        } else Swal.fire({ icon: 'error', title: 'Oops...', text: res.message });
      }).withFailureHandler(err => {
        btnSubmit.innerHTML = originalText; btnSubmit.disabled = false; Swal.fire({ icon: 'error', title: 'Gagal', text: 'Terjadi kesalahan pada Server.' });
      }).simpanAtauUpdateData(data);
    });
